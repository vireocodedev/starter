## Provider Setup Patterns

### Minimal Setup

Only theming (required for all components):

```tsx
import { createTheme } from "@mui/material/styles";
import { RgoThemeProvider } from "@vireocodedev/starter-ui";

const theme = createTheme({
  palette: {
    primary: { main: "#1976d2" },
    secondary: { main: "#dc004e" },
  },
});

function App() {
  return (
    <RgoThemeProvider theme={theme}>
      <YourAppContent />
    </RgoThemeProvider>
  );
}
```

### Full Provider Stack

All providers in the correct nesting order:

```tsx
import {
  RgoThemeProvider,
  RgoQueryClientProvider,
  RgoLocalizationProvider,
  RgoConfirmProvider,
  RgoSnackbarProvider,
} from "@vireocodedev/starter-ui";

function App() {
  return (
    <RgoThemeProvider theme={theme}>
      <RgoQueryClientProvider>
        <RgoLocalizationProvider>
          <RgoConfirmProvider>
            <RgoSnackbarProvider>
              <YourAppContent />
            </RgoSnackbarProvider>
          </RgoConfirmProvider>
        </RgoLocalizationProvider>
      </RgoQueryClientProvider>
    </RgoThemeProvider>
  );
}
```

### Conditional Providers

Add only the providers you need:

```tsx
// With notifications only
function AppWithNotifications() {
  return (
    <RgoThemeProvider theme={theme}>
      <RgoSnackbarProvider>
        <YourAppContent />
      </RgoSnackbarProvider>
    </RgoThemeProvider>
  );
}

// With confirmations only
function AppWithConfirmations() {
  return (
    <RgoThemeProvider theme={theme}>
      <RgoConfirmProvider>
        <YourAppContent />
      </RgoConfirmProvider>
    </RgoThemeProvider>
  );
}
```

## Form Patterns

### Basic Form with Validation

Using RgoForm with Zod schema validation:

```tsx
import { RgoForm, RgoInputNumber, useRgoForm } from "@vireocodedev/starter-ui";
import { TextField } from "@mui/material";
import { Controller } from "react-hook-form";
import z from "zod";

const UserSchema = z.object({
  name: z.string().min(1, "Name is required"),
  email: z.string().email("Invalid email"),
  age: z.number().min(1).max(120),
});

type UserData = z.infer<typeof UserSchema>;

function UserForm() {
  const form = useRgoForm<UserData>({
    schema: UserSchema,
    defaultValues: { name: "", email: "", age: 0 },
  });

  return (
    <RgoForm form={form} onSubmit={data => console.log("Submitted:", data)} submitButtonText="Save User">
      <Controller
        name="name"
        control={form.control}
        render={({ field, fieldState }) => (
          <TextField
            {...field}
            label="Name"
            error={!!fieldState.error}
            helperText={fieldState.error?.message}
            fullWidth
          />
        )}
      />

      <Controller
        name="email"
        control={form.control}
        render={({ field, fieldState }) => (
          <TextField
            {...field}
            label="Email"
            type="email"
            error={!!fieldState.error}
            helperText={fieldState.error?.message}
            fullWidth
          />
        )}
      />

      <Controller
        name="age"
        control={form.control}
        render={({ field, fieldState }) => (
          <RgoInputNumber
            {...field}
            label="Age"
            error={!!fieldState.error}
            helperText={fieldState.error?.message}
            min={1}
            max={120}
            fullWidth
          />
        )}
      />
    </RgoForm>
  );
}
```

### Form with Multiple Input Types

Combining different RGO input components:

```tsx
import {
  RgoForm,
  RgoInputDate,
  RgoInputSelect,
  RgoInputPassword,
  RgoInputAutocomplete,
  useRgoForm,
} from "@vireocodedev/starter-ui";

function ComplexForm() {
  const form = useRgoForm({
    schema: ComplexFormSchema,
    defaultValues: {
      password: "",
      birthDate: null,
      country: "",
      skills: [],
    },
  });

  return (
    <RgoForm form={form} onSubmit={handleSubmit}>
      <RgoInputPassword name="password" label="Password" form={form} fullWidth />

      <RgoInputDate name="birthDate" label="Birth Date" form={form} fullWidth />

      <RgoInputSelect name="country" label="Country" options={["US", "UK", "CA", "DE"]} form={form} fullWidth />

      <RgoInputAutocomplete
        name="skills"
        label="Skills"
        options={["React", "TypeScript", "Node.js", "Python"]}
        multiple
        form={form}
        fullWidth
      />
    </RgoForm>
  );
}
```

## Table Patterns

### Client-Side Table with Features

Full-featured data table with pagination and sorting:

```tsx
import { RgoClientTable } from "@vireocodedev/starter-ui";

const columns = [
  { field: "name", headerName: "Name", sortable: true },
  { field: "email", headerName: "Email", sortable: true },
  { field: "role", headerName: "Role", sortable: false },
  {
    field: "actions",
    headerName: "Actions",
    sortable: false,
    renderCell: user => (
      <IconButton onClick={() => editUser(user.id)}>
        <EditIcon />
      </IconButton>
    ),
  },
];

function UserTable({ users }) {
  return (
    <RgoClientTable data={users} columns={columns} enablePagination enableSorting initialPageSize={10} stickyHeader />
  );
}
```

### Server-Side Table

Table with server-side pagination and sorting:

```tsx
import { RgoServerTable } from "@vireocodedev/starter-ui";

function ServerUserTable() {
  const [pagination, setPagination] = useState({
    page: 0,
    rowsPerPage: 10,
    sortBy: null,
    sortDirection: "asc",
  });

  const { data, isLoading } = useQuery({
    queryKey: ["users", pagination],
    queryFn: () => fetchUsers(pagination),
  });

  return (
    <RgoServerTable
      data={data?.users || []}
      columns={columns}
      loading={isLoading}
      totalCount={data?.total || 0}
      pagination={pagination}
      onPaginationChange={setPagination}
    />
  );
}
```

## Hook Usage Patterns

### Confirmation Dialogs

Using useRgoConfirm for user confirmations:

```tsx
import { useRgoConfirm } from "@vireocodedev/starter-ui";

function DeleteButton({ userId }) {
  const confirm = useRgoConfirm();

  const handleDelete = async () => {
    const confirmed = await confirm({
      title: "Delete User",
      message: "Are you sure you want to delete this user? This action cannot be undone.",
      confirmText: "Delete",
      cancelText: "Cancel",
      color: "error",
    });

    if (confirmed) {
      await deleteUser(userId);
    }
  };

  return (
    <IconButton onClick={handleDelete} color="error">
      <DeleteIcon />
    </IconButton>
  );
}
```

### Notifications

Using useSnackbar for user feedback:

```tsx
import { useSnackbar } from "@vireocodedev/starter-ui";

function SaveButton({ data }) {
  const snackbar = useSnackbar();

  const handleSave = async () => {
    try {
      await saveData(data);
      snackbar.success("Data saved successfully!");
    } catch (error) {
      snackbar.error("Failed to save data. Please try again.");
    }
  };

  return (
    <Button variant="contained" onClick={handleSave}>
      Save
    </Button>
  );
}
```

### Translations

Using useTranslation for internationalization:

```tsx
import { useTranslation } from "@vireocodedev/starter-ui";

function WelcomeMessage() {
  const t = useTranslation();

  return (

      <h1>{t("welcome.title")}</h1>
      <p>{t("welcome.description")}</p>
      <button>{t("common.getStarted")}</button>
    </div>
  );
}
```

## Component Composition Patterns

### Data Display with Loading States

Combining loader with data components:

```tsx
import { RgoQueryErrorLoaderSuspense, RgoValueWithUnit } from "@vireocodedev/starter-ui";

function MetricDisplay({ metricId }) {
  const { data, isLoading, error } = useQuery({
    queryKey: ["metric", metricId],
    queryFn: () => fetchMetric(metricId),
  });

  return (
    <RgoQueryErrorLoaderSuspense isLoading={isLoading} error={error}>
      <RgoValueWithUnit value={data?.value} unit={data?.unit} decimalPlaces={2} />
    </RgoQueryErrorLoaderSuspense>
  );
}
```

### Form Fields with Labels

Using RgoLabelBox for consistent form field styling:

```tsx
import { RgoLabelBox } from "@vireocodedev/starter-ui";

function FormField({ label, required, helperText, children }) {
  return (
    <RgoLabelBox label={label} required={required} helperText={helperText}>
      {children}
    </RgoLabelBox>
  );
}

// Usage
<FormField label="User Name" required helperText="Enter your full name">
  <TextField fullWidth />
</FormField>;
```

## Theme Customization Patterns

### Custom Theme with Brand Colors

Creating a branded theme:

```tsx
import { createTheme } from "@mui/material/styles";

const brandTheme = createTheme({
  palette: {
    primary: {
      main: "#2563eb", // Brand blue
      light: "#60a5fa",
      dark: "#1d4ed8",
    },
    secondary: {
      main: "#dc2626", // Brand red
      light: "#f87171",
      dark: "#b91c1c",
    },
    background: {
      default: "#f8fafc",
      paper: "#ffffff",
    },
  },
  typography: {
    fontFamily: '"Inter", "Roboto", "Helvetica", "Arial", sans-serif',
    h1: {
      fontWeight: 700,
    },
  },
  shape: {
    borderRadius: 8,
  },
});
```

### Dark Theme Support

Supporting light and dark themes:

```tsx
import { useTheme } from "@vireocodedev/starter-ui";

function ThemeToggleButton() {
  const { toggleTheme, isDarkMode } = useTheme();

  return (
    <IconButton onClick={toggleTheme} aria-label="Toggle theme">
      {isDarkMode ? <LightModeIcon /> : <DarkModeIcon />}
    </IconButton>
  );
}
```

These patterns provide a solid foundation for building robust applications with RGO Front UI. Use them as starting points and adapt them to your specific requirements.
